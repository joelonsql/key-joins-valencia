\begin{lrbox}{\rawon}\begin{minipage}{11.9cm}
\begin{lstlisting}[style=sql]
SELECT o.id,
       SUM(oi.amount) AS item_total,
       SUM(p.amount)  AS payment_total
FROM orders AS o
LEFT JOIN order_items AS oi ON oi.order_id = o.id
LEFT JOIN payments    AS p  ON p.order_id  = o.id
GROUP BY o.id;
\end{lstlisting}
\vspace{.25cm}
\begin{lstlisting}[style=mono]
 id  | item_total | payment_total
-----+------------+---------------
 100 |     200.00 |        200.00
 101 |      30.00 |         30.00
(2 rows)
\end{lstlisting}
\end{minipage}\end{lrbox}

\begin{lrbox}{\rawkey}\begin{minipage}{13.8cm}
\begin{lstlisting}[style=sql]
SELECT o.id,
       SUM(oi.amount) AS item_total,
       SUM(p.amount)  AS payment_total
FROM orders AS o
LEFT JOIN order_items AS oi FOR KEY o (id) <- oi (order_id)
LEFT JOIN payments    AS p  FOR KEY o (id) <- p (order_id)
GROUP BY o.id;
\end{lstlisting}

\begin{lrbox}{\groupedon}\begin{minipage}{11.9cm}

       oi.item_total,
       p.payment_total
FROM orders AS o

    SELECT order_id, SUM(amount) AS item_total
    FROM order_items
    GROUP BY order_id
) AS oi ON o.id = oi.order_id
LEFT JOIN (
    SELECT order_id, SUM(amount) AS payment_total
    FROM payments
    GROUP BY order_id
) AS p ON o.id = p.order_id;
